cmake_minimum_required(VERSION 3.20)
project(Eternity VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ODBC REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(eternity_storage src/storage/sql_server.cpp)
target_include_directories(eternity_storage PUBLIC include)
target_link_libraries(eternity_storage PUBLIC ODBC::ODBC)
target_compile_options(eternity_storage PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

add_executable(eternity-server src/main.cpp)
add_library(eternity_game src/game/definitions.cpp)
target_include_directories(eternity_game PUBLIC include)
target_link_libraries(eternity_game PUBLIC eternity_storage)
target_compile_options(eternity_game PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(eternity_auth src/auth/accounts.cpp)
target_include_directories(eternity_auth PUBLIC include)
target_link_libraries(eternity_auth PUBLIC eternity_storage)
target_compile_options(eternity_auth PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(eternity_net src/net/framing.cpp src/net/login_crypto.cpp)
target_include_directories(eternity_net PUBLIC include)
target_compile_options(eternity_net PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_link_libraries(eternity_net PUBLIC OpenSSL::Crypto)
target_link_libraries(eternity-server PRIVATE eternity_game eternity_auth eternity_net Threads::Threads)
target_compile_options(eternity-server PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

include(CTest)
if(BUILD_TESTING)
    add_executable(storage-unit tests/storage_unit.cpp)
    target_link_libraries(storage-unit PRIVATE eternity_storage)
    add_test(NAME storage-unit COMMAND storage-unit)
    add_executable(storage-integration tests/storage_integration.cpp)
    target_link_libraries(storage-integration PRIVATE eternity_storage eternity_auth eternity_game)
    add_executable(framing-unit tests/framing_unit.cpp)
    target_link_libraries(framing-unit PRIVATE eternity_net)
    add_test(NAME framing-unit COMMAND framing-unit)
    add_executable(login-crypto-unit tests/login_crypto_unit.cpp)
    target_link_libraries(login-crypto-unit PRIVATE eternity_net)
    add_test(NAME login-crypto-unit COMMAND login-crypto-unit)
endif()

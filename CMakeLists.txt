cmake_minimum_required(VERSION 3.20)
project(Eternity VERSION 0.1.0 LANGUAGES CXX)
option(ETERNITY_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ETERNITY_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ODBC REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
add_library(eternity_storage src/storage/sql_server.cpp)
target_include_directories(eternity_storage PUBLIC include)
target_link_libraries(eternity_storage PUBLIC ODBC::ODBC)
target_compile_options(eternity_storage PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

add_executable(eternity-server src/main.cpp)
add_library(eternity_game src/game/definitions.cpp src/game/characters.cpp src/game/creation.cpp src/game/item_ids.cpp src/game/character_commands.cpp)
target_include_directories(eternity_game PUBLIC include)
target_link_libraries(eternity_game PUBLIC eternity_storage Threads::Threads)
target_compile_options(eternity_game PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(eternity_auth src/auth/accounts.cpp src/auth/login_protocol.cpp src/auth/sql_authenticator.cpp)
target_include_directories(eternity_auth PUBLIC include)
target_link_libraries(eternity_auth PUBLIC eternity_storage)
target_compile_options(eternity_auth PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(eternity_net src/net/framing.cpp src/net/login_crypto.cpp src/net/world_crypto.cpp)
target_include_directories(eternity_net PUBLIC include)
target_compile_options(eternity_net PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_link_libraries(eternity_net PUBLIC OpenSSL::Crypto)
target_link_libraries(eternity_auth PUBLIC eternity_net)
add_library(eternity_session src/session/registry.cpp)
target_include_directories(eternity_session PUBLIC include)
target_link_libraries(eternity_session PUBLIC eternity_net Threads::Threads)
target_compile_options(eternity_session PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(eternity_world src/net/world_protocol.cpp src/net/world_server.cpp)
target_include_directories(eternity_world PUBLIC include ${ASIO_INCLUDE_DIR})
target_compile_definitions(eternity_world PUBLIC ASIO_STANDALONE)
target_link_libraries(eternity_world PUBLIC eternity_session eternity_game)
target_compile_options(eternity_world PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_library(eternity_login src/net/login_server.cpp)
target_include_directories(eternity_login PUBLIC include ${ASIO_INCLUDE_DIR})
target_compile_definitions(eternity_login PUBLIC ASIO_STANDALONE)
target_link_libraries(eternity_login PUBLIC eternity_auth eternity_session Threads::Threads)
target_compile_options(eternity_login PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_link_libraries(eternity-server PRIVATE eternity_game eternity_login eternity_world Threads::Threads)
target_compile_options(eternity-server PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
